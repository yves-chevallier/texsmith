% Chapter head pages. Two-sided (the class default): the folio sits on the
% outer edge; one-sided: on the right.

\fancypagestyle{plain}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \if@twoside
    \fancyfoot[LE,RO]{\thepage}
  \else
    \fancyfoot[RO]{\thepage}
  \fi
}

% Mainmatter regular page style: the chapter on the verso, the section on
% the recto. One-sided pages carry the section only — both marks on one
% line overflow a small page.
\fancypagestyle{fancy}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0.5pt}
  \if@twoside
    \fancyhead[LE]{\scshape \leftmark}
    \fancyhead[RO]{\scshape \rightmark}
    \fancyfoot[LE,RO]{\thepage}
  \else
    \fancyhead[R]{\scshape \rightmark}
    \fancyfoot[R]{\thepage}
  \fi
}

% this is not very readable. This command changes the style to display
% the description text on the next line.

  itemindent=\dimexpr-5pt-\labelsep\relax
}
