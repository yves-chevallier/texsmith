\section{API Sections}\label{api-sections}

\begin{description}
\item[{ \hyperref[snippet:<HASH>]{\texttt{high-\allowbreak{}level}} }] High-level orchestration helpers (\texttt{ConversionService}, \texttt{TemplateSession}) for programmatic conversions and template sessions.
\item[{ \hyperref[snippet:<HASH>]{\texttt{core}} }] Core package modules (\texttt{texsmith}, configuration, contexts, conversion helpers, etc.).
\item[{ \hyperref[snippet:<HASH>]{\texttt{bibliography}} }] Bibliography tooling (BibTeX parsing, DOI resolution, issue reporting).
\item[{ \hyperref[snippet:<HASH>]{\texttt{cli}} }] Command-line entry points and utilities.
\item[{ \hyperref[snippet:<HASH>]{\texttt{handlers}} }] Readers and writers of the typed IR pipeline (\texttt{read(HTML) \(\rightarrow\) IR \(\rightarrow\) write(IR) \(\rightarrow\) LaTeX}), with the \texttt{@reads} / \texttt{@writes} decorators.
\item[{ \hyperref[snippet:<HASH>]{\texttt{latex}} }] LaTeX infrastructure (formatter, renderer, templates).
\item[{ \hyperref[snippet:<HASH>]{\texttt{markdown}} }] Markdown conversion helpers and custom extensions.
\item[{ \hyperref[snippet:<HASH>]{\texttt{plugins}} }] Optional integrations (MkDocs Material-specific handlers).
\item[{ \hyperref[snippet:<HASH>]{\texttt{transformers}} }] Asset conversion strategies (SVG, Draw.io, Mermaid, remote images).

\end{description}
